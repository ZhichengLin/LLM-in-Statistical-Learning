theme_figure5 <- function() {
  theme_classic(base_family = FONT_FAMILY, base_size = FONT_SIZE_PT) +
    theme(
      text = element_text(family = FONT_FAMILY, size = FONT_SIZE_PT),
      axis.line = element_line(colour = "#000000", linewidth = AXIS_LINE_SIZE),
      axis.ticks = element_line(colour = "#000000", linewidth = AXIS_LINE_SIZE),
      axis.ticks.length = grid::unit(TICK_LENGTH_PT, "pt"),
      axis.text = element_text(colour = "black", size = FONT_SIZE_PT),
      axis.text.x = element_text(margin = margin(t = 3)),
      axis.title = element_text(colour = "black", size = FONT_SIZE_PT),
      plot.title = element_text(
        face = "bold", hjust = 0.5, size = FONT_SIZE_PT,
        margin = margin(t = 7, b = 1)
      ),
      plot.subtitle = element_text(
        hjust = 0.5, size = FONT_SIZE_PT,
        margin = margin(b = 3)
      ),
      legend.title = element_text(size = FONT_SIZE_PT),
      legend.text = element_text(size = FONT_SIZE_PT),
      legend.key = element_blank(),
      legend.background = element_blank(),
      legend.key.width = grid::unit(6, "pt"),
      legend.key.height = grid::unit(7, "pt"),
      legend.spacing.y = grid::unit(0.5, "pt"),
      legend.margin = margin(0, 0, 0, 0),
      strip.background = element_blank(),
      strip.text = element_text(size = FONT_SIZE_PT),
      panel.grid = element_blank(),
      plot.margin = margin(4, 4, 4, 4)
    )
}
