summarise_mean_ci <- function(data, value, group_vars) {
  data %>%
    group_by(across(all_of(group_vars))) %>%
    summarise(
      mean = mean(.data[[value]], na.rm = TRUE),
      sd = sd(.data[[value]], na.rm = TRUE),
      n = sum(!is.na(.data[[value]])),
      .groups = "drop"
    ) %>%
    mutate(
      se = sd / sqrt(n),
      ci95 = qt(0.975, df = pmax(n - 1, 1)) * se
    )
}

make_rating_bundle <- function(stage2_col, stage3_col) {
  all_long <- data_filtered %>%
    select(ID, Label, all_of(c(stage2_col, stage3_col)))
  names(all_long) <- c("ID", "Label", "Stage 2", "Stage 3")

  all_long <- all_long %>%
    pivot_longer(
      cols = c("Stage 2", "Stage 3"),
      names_to = "Phase",
      values_to = "Rating"
    ) %>%
    drop_na(Rating) %>%
    mutate(
      Label = factor(Label, levels = GROUP_LEVELS),
      Phase = factor(Phase, levels = c("Stage 2", "Stage 3"))
    )

  all_summary <- summarise_mean_ci(
    all_long,
    value = "Rating",
    group_vars = c("Label", "Phase")
  )

  completer_long <- data_filtered %>%
    filter(followup_complete) %>%
    select(ID, Label, all_of(c(stage2_col, stage3_col)))
  names(completer_long) <- c("ID", "Label", "Stage 2", "Stage 3")

  completer_long <- completer_long %>%
    pivot_longer(
      cols = c("Stage 2", "Stage 3"),
      names_to = "Phase",
      values_to = "Rating"
    ) %>%
    drop_na(Rating) %>%
    mutate(
      Label = factor(Label, levels = GROUP_LEVELS),
      Phase = factor(Phase, levels = c("Stage 2", "Stage 3"))
    )

  completer_summary <- summarise_mean_ci(
    completer_long,
    value = "Rating",
    group_vars = c("Label", "Phase")
  )

  list(
    raw = all_long,
    all = all_summary,
    completers = completer_summary
  )
}

format_p <- function(p) {
  if (length(p) != 1 || !is.finite(p)) {
    stop("A figure annotation did not resolve to one finite P value.")
  }

  if (p < 0.001) {
    "italic(P)~'< 0.001'"
  } else {
    sprintf("italic(P)~'= %.3f'", p)
  }
}

extract_full_sample_p <- function(measure) {
  result_path <- paste0(
    "../Study_2_Main_Results/results/H2/",
    "H2_Within_Task_Attitude_Perceptions.xlsx"
  )
  sheet <- paste0(measure, "_Posthoc_Stage_Label")

  result <- read_excel(result_path, sheet = sheet) %>%
    filter(
      .data$contrast == "stage3 - stage2",
      .data$Label %in% GROUP_LEVELS
    ) %>%
    mutate(Label = factor(.data$Label, levels = GROUP_LEVELS)) %>%
    arrange(.data$Label)

  if (nrow(result) != length(GROUP_LEVELS) || any(!is.finite(result$p.value))) {
    stop(sprintf(
      "Could not recover the three %s P values from %s [%s].",
      measure, result_path, sheet
    ))
  }

  setNames(result$p.value, as.character(result$Label))
}

fit_completer_p <- function(measure) {
  completer_long <- data_filtered %>%
    filter(.data$followup_complete) %>%
    mutate(
      Label = factor(.data$Label, levels = GROUP_LEVELS),
      Used_LLM_Hours = case_when(
        .data$Used_LLM_Hours == "0" ~ "None",
        .data$Used_LLM_Hours %in% c("1–5", "6–10") ~ "Low",
        .data$Used_LLM_Hours %in% c("11–20", ">20") ~ "High",
        TRUE ~ NA_character_
      ),
      Used_LLM_Hours = droplevels(factor(
        .data$Used_LLM_Hours,
        levels = c("None", "Low", "High")
      ))
    ) %>%
    select(
      ID, Label, Age, Gender, Field, Used_LLM_Hours,
      matches(paste0("^stage[23]_task[1-3]_", measure, "$"))
    ) %>%
    pivot_longer(
      cols = matches("^stage[23]_task[1-3]_"),
      names_to = "fullvar",
      values_to = "Value"
    ) %>%
    mutate(
      Phase = str_extract(.data$fullvar, "^stage[23]"),
      Task = str_extract(.data$fullvar, "task[1-3]"),
      Phase = factor(.data$Phase, levels = c("stage3", "stage2")),
      ID = factor(.data$ID),
      Task = factor(.data$Task)
    )

  model <- lmer(
    Value ~ Phase * Label + Age + Gender + Field + Used_LLM_Hours +
      (1 | ID) + (1 | Task) + (1 | ID:Task),
    data = completer_long
  )

  result <- as.data.frame(
    pairs(emmeans(model, ~ Phase | Label))
  ) %>%
    filter(
      .data$contrast == "stage3 - stage2",
      .data$Label %in% GROUP_LEVELS
    ) %>%
    mutate(Label = factor(.data$Label, levels = GROUP_LEVELS)) %>%
    arrange(.data$Label)

  if (nrow(result) != length(GROUP_LEVELS) || any(!is.finite(result$p.value))) {
    stop(sprintf(
      "Could not calculate the three completer-only %s P values.",
      measure
    ))
  }

  setNames(result$p.value, as.character(result$Label))
}
